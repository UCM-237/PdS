\begin{frame}
\frametitle{Valor medio y Esperanza matemática}

% \vspace{-0.2cm}
  \begin{itemize}
    \item Podemos calcular el valor medio de dos formas equivalentes:
    \begin{itemize} 
      \item $\frac{1}{N}\sum_{i}x_i=\sum_{x_i\in X}f_i x_i$ $f_i=\frac{ocurrencias~x_i}{N}$
      \item \emph{Ejemplo:} $X = \{1, 1, 3, 2, 5\}$
      \begin{itemize}
        \item $E\{X\}=\frac{(1+1+3+2+5)}{5}=2.4$
        \item $E\{X\}=\frac{2}{5}1+\frac{1}{5}2+\frac{1}{5}3+\frac{1}{5}5=2.4$
      \end{itemize}
      \item Interpretar cada término como $f_i x_i$
    \end{itemize}
    \item Esperanza Matemática: $E\{X\}$
    \begin{itemize}
      \item Generaliza el concepto de valor medio.
      \item Discreto: $E\{X\}=\sum_{x_i\in X}g(x_i)p_{x}(x_i)$
      \item Continuo: $E\{X\}=\int_{-\infty}^{\infty}g(x)p_{x}(x)dx$
      \item Valor esperado de una función genérica: $g(x)$
    \end{itemize}
  \end{itemize}

\begin{frame}
  \frametitle{Estadísticos}
  \begin{itemize}
    \item Un estadístico es una medida cuantitiva de una muestra.
    \item Trata de inferir o estimar una propiedad de la población.
    \item Estadísticos frecuentes:
    \begin{itemize}
      \item $\mu_x=E\{X\}$ (media)
      \item $\sigma_x^2=E\{(X-\mu_x)^2\}$ (varianza)
      \item $E\{X^2\}=\sigma_x^2+\mu_x^2$ (valor cuadrático medio)
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Muestreo variable aleatoria}
  \begin{itemize}
    \item Una variable aleatoria modela un experimento aleatorio.
    \item Podemos utilizarlo desde dos enfoques complementarios: Muestreo $\equiv$ Caracterización
    \item Muestreo: generar muestras que se ajusten a la descripción de X
    \item Caracterización: utilizar los resultados para ajustar la descripción de X 
  \end{itemize}
\end{frame}
