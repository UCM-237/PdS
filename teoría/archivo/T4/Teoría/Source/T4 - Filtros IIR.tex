\begin{frame}
  \frametitle{Transformación del espacio de trabajo ($s \leftrightarrow z$)}
  \centering Filtro continuo $H_a(s)$ $\rightarrow$  filtro discreto: $H(z)=H_a(s)|_{s=F(z)}$
  
  \vspace{0.4cm}
  \begin{itemize}
    \footnotesize
    \item \alert{Transformada bilineal: $\:\:s=\frac{2}{T}\left ( \frac{1-z^{-1}}{1+z^{-1}}\right) \: \: \leftrightarrow \:\: z=\frac{1+T\cdot s/2}{1-T\cdot s/2}$}

    \begin{columns}[T]
      \hspace{1cm}\begin{column}{7.5cm}
        \begin{itemize}
          \vspace{0.1cm}
          \scriptsize
          % \item Se obtiene mediante la aplicación del método de integración numérica trapezoidal a la representación de la ecuación diferencial correspondiente a $H_a(s)$. T es el periodo del paso de integración.
          \item El semiplano izquierdo $s$ va al círculo unidad en $z$.
          \item Se puede utilizar para todos los filtros.
          \item Relación en frecuencias: \alert{$\omega_a=\frac{2}{T}tan(\frac{\omega_d}{2})$}.
        \end{itemize}
      \end{column}
      \begin{column}{4cm}
        \includegraphics[trim=0cm 0cm 0cm 0cm, clip=true,angle=-90,width=2.5cm]{Figures/Bilineal.png}
      \end{column}
    \end{columns}

    \begin{columns}[T]
      \hspace{0.4cm}\begin{column}{11cm}
      \begin{itemize}
        \scriptsize
        \item Relación entre las frecuencias continuas ($s=j\omega_a$) y discretas ($z=e^{j\omega_d}$)

        $j\omega_a=\frac{2}{T}\left ( \frac{1-e^{-j\omega_d}}{1+e^{-j\omega_d}}\right)=\frac{2}{T}\left ( \frac{e^{j\omega_d/2}-e^{-j\omega_d/2}}{e^{j\omega_d/2}+e^{-j\omega_d/2}}\right)=\frac{2}{T} j \frac{sen(\omega_d/2)}{cos(\omega_d/2)}=j\frac{2}{T} tan(\frac{\omega_d}{2})$

        Se cumple que  $\omega_d \!\in \![0,\pi] \leftrightarrow \omega_a \!\in\! [0,\infty)$; $\omega_d  \!\in\! [0,-\pi] \leftrightarrow \omega_a \!\in\! [0,-\infty)$
        \color{black}
        % \item Se produce un \emph{combado} de las frecuencias, asociado al comportamiento no lineal de la tangente.
        \item \alert{Si hacemos T=2, las expresiones se simplifican.}
      \end{itemize}
      \end{column}
    \end{columns}

  \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Diseño de filtros IIR paso bajo}
\small
\begin{enumerate}
  \item Especificación del filtro deseado ($\color{blue}\omega_p,\alpha_p,\omega_s,\alpha_s$).
  \item Precombado de frecuencias, $\color{blue}\omega_a=\frac{2}{T}tan(\omega_d/2)$.
  \item Diseñar el filtro continuo $\color{blue}H_a(s)$.
  \item Discretizar el filtro con la transformada bilineal, $\color{blue}s=\frac{2}{T}\left ( \frac{1-z^{-1}}{1+z^{-1}}\right)$ %para obtener la función de transferencia del filtro discreto.
  \begin{itemize}
  \item Con Matlab: \alert{\texttt{[numz,denz]=bilinear(nums,dens,1/T)}}.
  \end{itemize}

\end{itemize}

\end{enumerate}

\begin{frame}
\frametitle{Filtros  IIR paso alto, paso banda, supresor de banda}
\small
\vspace{-0.4cm}
  \begin{enumerate}
  \item Precombado de frecuencias ($\omega_a=\frac{2}{T}tan(\omega_d/2)$).% para obtener la especificación en frecuencias del filtro continuo $H_a(s)$.
  \vspace{-0.1cm}
  \item Convertir las especificaciones de $H_a(s)$ en paso bajo $H_{LP}(s)$.
  \vspace{0.1cm}
  \begin{columns}[T]
    \hspace{0.8cm}\begin{column}{2cm}
    \scriptsize
    Paso alto:

    \vspace{0.2cm}

    $s=\frac{\omega_p \hat \omega_p}{\hat s}$

    \vspace{0.2cm}

    $- \omega \hat \omega=\omega_p \hat \omega_p$
    \end{column}
    \hspace{-0.8cm}\begin{column}{3cm}
    \scriptsize
    Paso banda:

    \vspace{0.1cm}

    $s=\omega_p\frac{\hat s^2+\hat \omega_0^2}{\hat s (\hat \omega_{p2}-\hat \omega_{p1})}$

    $\omega=\omega_p\frac{\hat \omega^2-\hat \omega_0^2}{\hat \omega (\hat \omega_{p2}-\hat \omega_{p1})}$

    $\hat \omega_0^2=\hat \omega_{p1}\hat \omega_{p2}=\hat \omega_{s1}\hat \omega_{s2}$

    \end{column}
    \hspace{-0.8cm}\begin{column}{3cm}
    \scriptsize
    Supresor de banda:

    \vspace{0.1cm}

    $s=\omega_s\frac{\hat s (\hat \omega_{s2}-\hat \omega_{s1})}{\hat s^2+\hat \omega_0^2}$

    $\omega=\omega_s\frac{\hat \omega (\hat \omega_{s2}-\hat \omega_{s1})}{\hat \omega_0^2-\hat \omega^2}$

    $\hat \omega_0^2=\hat \omega_{p1}\hat \omega_{p2}=\hat \omega_{s1}\hat \omega_{s2}$

    \end{column}
  \end{columns}

  \item Diseñar el filtro continuo paso bajas $H_{LP}(s)$
  
  \item Convertir el filtro paso baja $H_{LP}(s)$ en el filtro destino $H_a(s)$.

  \item Discretizar $H_a(s)$ con la bilineal ($s=\frac{2}{T}\left ( \frac{1-z^{-1}}{1+z^{-1}}\right)|_{T=2}$)
  \[
    H(z) = H_a\left(\frac{1-z^{-1}}{1+z^{-1}}\right)
  \]

  \end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Relaciones espectrales discretas}

\begin{itemize}
\footnotesize
\item La transformación espectral discreta $z=F(\hat z)$ permite obtener a partir de un filtro discreto paso bajo $H_{LP}(z)$  otro filtro discreto$^1$ (paso alto, paso banda, supresor de banda) $H(\hat z)=H_{LP}(z)|_{z=F(\hat z)}$.
\begin{itemize}
\footnotesize

\item Filtro paso alta: $z^{-1}=-\frac{\hat z^{-1}+\lambda}{1+\lambda \hat z^{-1}}$, con
$\lambda=-\frac{cos((w_c-	\hat w_c)/2)}{cos((w_c+	\hat w_c)/2)}$



\item Filtro paso banda: $z^{-1}=-\frac{\hat z^{-2}-\frac{2\lambda \rho}{\rho+1}\hat z^{-1}+\frac{\rho-1}{\rho+1}}{\frac{\rho-1}{\rho+1}\hat z^{-2}-\frac{2\lambda \rho}{\rho+1}\hat z^{-1}+1}$,

$\lambda=\frac{cos((\hat w_{c2}+	\hat w_{c1})/2)}{cos((\hat w_{c2}-	\hat w_{c1})/2)}$, $\rho=\cot((\hat w_{c2}-\hat w_{c1})/2)\tan(w_c/2)$

\item Filtro supresor banda: $z^{-1}=\frac{\hat z^{-2}-\frac{2\lambda}{\rho+1}\hat z^{-1}+\frac{1-\rho}{\rho+1}}{\frac{1-\rho}{\rho+1}\hat z^{-2}-\frac{2\lambda}{\rho+1}\hat z^{-1}+1}$,

$\lambda=\frac{cos((\hat w_{c2}+\hat w_{c1})/2)}{cos((\hat w_{c2}-\hat w_{c1})/2)}$, $\rho=\tan((\hat w_{c2}-\hat w_{c2})/2)\tan(w_c/2)$


\end{itemize}

\item Con Matlab$^{2}$
\begin{itemize}
\footnotesize
\item \texttt{[npa,dpa]=iirlp2hp(numlp,denlp,wplp,wphp)}
\item \texttt{[npa,dpa]=iirlp2bp(numlp,denlp,wplp,[wp1d,wp2d])}
\item \texttt{[nbs,dbs]=iirlp2sp(numlp,denlp,wslp,[ws1d,ws2d])}
\end{itemize}
\end{itemize}

\vspace{-0.2cm}
\rule{10cm}{0.4pt}

\scriptsize
$^1$También existe una relación espectral que permite transformar un filtro paso bajo en otro paso bajo, que se utiliza para cambiar las especificaciones en frecuencia de un filtro paso baja ya diseñado.

$^2$ Las frecuencias tienen que estar  normalizadas entre $[0,1]$ que se corresponden a los valores discretos entre $[0,\pi]$


\end{frame}
