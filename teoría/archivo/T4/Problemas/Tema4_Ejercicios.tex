		\itemP Para cada señal, calcula y representa su DFT de $N$ muestras en MATLAB (mediante \color{blue}\texttt{fft}\color{black}). Indica el valor de $N$ y del periodo de muestreo $T_s$ ($t=n T_s$) y justifica su elección.
			\begin{itemize}
				\itemB $x(t)=\textrm{sin}(\frac{\pi}{2}t)$
				\itemB $x(t)=\sum_{k=0}^{10}0.9^k sin(\frac{k\pi}{2}t)$
				\itemB $x_1(t)=\textrm{e}^{-t}, 0\leq t<5$
				\itemB $x_2(t)=\textrm{sinc}(t)$
				\itemB $x_2(t)=\textrm{randn}(N,1)$
			\end{itemize}

		\itemP Reconstruye las señales del ejercicio 1 a partir de la DFT, en los casos en que sea posible.

		\itemP Reconstruye la DTFT de las señales del ejercicio 1 a partir de la DFT, en los casos en que sea posible.
		
		\itemP Reconstruye la transformada Z de las señales del ejercicio 1 a partir de la DFT, en los casos en que sea posible.

		\itemP Calcula, mediante la DFT, la respuesta de los siguientes sistemas LTI tomando como entrada las señales del ejercicio 1. 
		\begin{itemize}
			\itemB $h(n)=\frac{1}{N}, 0\leq n < N=20$
			\itemB $h(n)=0.7^{-n}$
			\itemB $H(z)=0.8 + 0.5z^{-1} - 0.2 z^{-2} - 4 z^{-3} + z^{-4}$
			\itemB $H(z)=\frac{0.8}{1 + 0.5z^{-1}}$
		\end{itemize}
