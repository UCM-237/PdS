\section*{Señales}

\begin{ejercicio}[clasificación]
  Clasifica las siguientes señales según su dimensión, su número de canales y si la variable independiente y la amplitud son continuas o discretas.
  \begin{incisos}[2]
    \item $x(t)=\begin{pmatrix}4t\\ \e^{j2t}\end{pmatrix}$
    \item $x[n]=4\,\e^{j\frac35n}$
    \item $x[n]=3\,\e^{j\frac2\pi n}$
    \item $x[n]=\big\lfloor\sin\big(\tfrac{2\pi n}{20}\big)\big\rfloor$
    \item $x(t,r)=\sin(\omega t-\kappa r)$, donde $r$ es una posición en metros
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item Unidimensional (depende solo de $t$), vectorial de dos canales, con tiempo y amplitud continuos: analógica. El segundo canal es complejo.
    \item Unidimensional, escalar y compleja; tiempo discreto y amplitud continua: es una secuencia muestreada (no cuantificada).
    \item Igual que la anterior.
    \item Unidimensional y escalar, con tiempo discreto y amplitud discreta: solo toma los valores $\{-1,0,1\}$ (vale $1$ cuando el seno vale exactamente $1$, en $n=5+20l$). Es una señal digital.
    \item Bidimensional (tiempo y espacio), escalar y real, con ambas variables y la amplitud continuas. Es una onda que viaja en el sentido de $r$ creciente a velocidad $\omega/\kappa$.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[periodicidad y paridad]
  Clasifica las siguientes señales según su periodicidad y su paridad. Para las periódicas, indica su periodo fundamental.
  \begin{incisos}[3]
    \item $x(t)=4\,\e^{j\frac35t}$
    \item $x(t)=3t$
    \item $x(t)=3t^2$
    \item $x[n]=\sin(3n)$
    \item $x[n]=\cos\big(\tfrac{\pi n}{20}\big)$
    \item $x[n]=\cos(3n-4)\,\big(u[n]-u[n-N_0-1]\big)$, con $N_0>0$ entero
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item Periódica con $T_0=2\pi/(3/5)=10\pi/3$. No es par ni impar, pero es hermítica: $x^*(-t)=x(t)$, con parte real $4\cos(3t/5)$ par e imaginaria $4\sin(3t/5)$ impar.
    \item Aperiódica e impar.
    \item Aperiódica y par.
    \item $F_0=3/(2\pi)$ es irracional: \emph{no} es periódica, aunque $\sin(3t)$ sí lo sea. Es impar.
    \item $F_0=1/40$: periódica con $N_0=40$. Es par.
    \item Tiene duración finita (solo es no nula en $0\le n\le N_0$), así que no es periódica. No es par ni impar: $x[-1]=0$ pero $x[1]=\cos(-1)\ne0$, y $x[0]=\cos(-4)\ne0$.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[energía y potencia]
  Calcula la energía y la potencia media de las siguientes señales e indica si son de energía, de potencia o de ninguno de los dos tipos.
  \begin{incisos}[2]
    \item $x(t)=t$
    \item $x(t)=u(t)$
    \item $x(t)=\e^{-j4t}$
    \item $x(t)=4t$ para $0\le t\le10$ y $0$ en otro caso
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $E=\infty$ y $\frac{1}{2T}\int_{-T}^{T}t^2\dd t=T^2/3\to\infty$: $P=\infty$. No es de energía ni de potencia.
    \item $E=\infty$ y $P=\lim\frac{1}{2T}\int_0^T\dd t=\tfrac12$: señal de potencia.
    \item $\abs{x(t)}=1$, así que $E=\infty$ y $P=1$: señal de potencia.
    \item $E=\int_0^{10}16t^2\dd t=16000/3\approx5333$ y $P=0$: señal de energía.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[señales complejas]
  Indica si las siguientes señales son reales o complejas y, en este caso, si son hermíticas.
  \begin{incisos}[3]
    \item $x(t)=\e^{-3t}$
    \item $x(t)=\e^{j2\pi t}$
    \item $x[n]=\e^{j\pi n}$
    \item $x(t)=\cos(4t)+j\sin(5t)$
    \item $x(t)=\e^{j\pi}\,\e^{-j3t}$
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item Real.
    \item Compleja y hermítica: $x^*(-t)=\big(\e^{-j2\pi t}\big)^*=\e^{j2\pi t}=x(t)$.
    \item Real, aunque no lo parezca: $\e^{j\pi n}=\cos(\pi n)=(-1)^n$, porque $\sin(\pi n)=0$ para todo entero $n$.
    \item Compleja. Su parte real, $\cos(4t)$, es par y su parte imaginaria, $\sin(5t)$, es impar: es hermítica.
    \item Compleja: $\e^{j\pi}=-1$ y $x(t)=-\e^{-j3t}$. Es hermítica: $x^*(-t)=-\big(\e^{j3t}\big)^*=-\e^{-j3t}=x(t)$.
  \end{enumerate}
\end{solucion}

\section*{Operaciones y análisis en frecuencia}

\begin{ejercicio}[operaciones]
  Sean $x_1(t)=\sin t$, \ $x_2(t)=\cos\big(\tfrac{2\pi}{5}t\big)$, \ $x_3[n]=\{\underline{0},3,1,1\}$ (el subrayado marca $n=0$) y $x_4[n]=\e^{j4n}$. Obtén:
  \begin{enumerate}[label=\alph*)]
    \item el producto $x_1(t)\,x_2(t)$ y la suma $x_1(t)+x_2(t)$; ¿es periódica la suma?
    \item la convolución $(x_1\conv x_2)(t)$;
    \item $x_3[n]/x_4[n]$ y $x_3[n]-x_4[n]$;
    \item la convolución $(x_3\conv x_4)[n]$;
    \item $x_3[n-2]$, \ $x_3[-n]$, \ el diezmado $x_3[2n]$ y la expansión por $L=2$ de $x_3$.
  \end{enumerate}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $x_1x_2=\sin t\cos\big(\tfrac{2\pi}{5}t\big)=\tfrac12\Big[\sin\big((1+\tfrac{2\pi}{5})t\big)+\sin\big((1-\tfrac{2\pi}{5})t\big)\Big]$. La suma $\sin t+\cos(2\pi t/5)$ no es periódica: los periodos son $2\pi$ y $5$, y su cociente $2\pi/5$ es irracional.
    \item La integral $\int\sin\tau\cos\big(\tfrac{2\pi}{5}(t-\tau)\big)\dd\tau$ no converge. Interpretada mediante la transformada de Fourier, el resultado es nulo: los espectros son deltas en $\omega=\pm1$ y $\omega=\pm2\pi/5$, que no coinciden, así que su producto es cero. Un sistema LTI con $h=x_2$ elimina la frecuencia $\omega=1$.
    \item $x_3[n]/x_4[n]=x_3[n]\,\e^{-j4n}=\{\underline{0},\,3\e^{-j4},\,\e^{-j8},\,\e^{-j12}\}$ y vale $0$ fuera de $0\le n\le3$. \ $x_3[n]-x_4[n]$ vale $x_3[n]-\e^{j4n}$ en $0\le n\le3$ y $-\e^{j4n}$ en el resto.
    \item $(x_3\conv x_4)[n]=\sum_kx_3[k]\,\e^{j4(n-k)}=\e^{j4n}\sum_kx_3[k]\,\e^{-j4k}=\big(3\e^{-j4}+\e^{-j8}+\e^{-j12}\big)\,\e^{j4n}$. La exponencial sale multiplicada por una constante: es una autofunción del sistema LTI con $h=x_3$, y la constante es $X_3(\e^{j4})$.
    \item $x_3[n-2]=\{\underline{0},0,0,3,1,1\}$ (no nula en $n=3,4,5$). \ $x_3[-n]=\{1,1,3,\underline{0}\}$ (de $n=-3$ a $n=0$). \ $x_3[2n]=\{\underline{0},1\}$: se conservan $x_3[0]$ y $x_3[2]$ y se pierden las demás. \ Expansión: $\{\underline{0},0,3,0,1,0,1\}$.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[convolución]
  Calcula las siguientes convoluciones.
  \begin{incisos}[2]
    \item $x[n]=\{\underline{1},4,2,0,1\}$ e $y[n]=\{\underline{4},2,3\}$
    \item $x[n]=4\,\delta[n-6]$ e $y[n]=\e^{j\frac{2\pi}{5}n}$
    \item $x[n]=u[n]$ e $y[n]=a^n u[n]$
    \item $x(t)=u(t)-u(t-2)$ e $y(t)=u(t)-u(t-1)$
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $(x\conv y)[n]=\{\underline{4},18,19,16,10,2,3\}$, con $5+3-1=7$ muestras desde $n=0$. Se comprueba con \texttt{np.convolve} o \texttt{conv}.
    \item Convolucionar con $\delta[n-6]$ desplaza: $4\,\e^{j\frac{2\pi}{5}(n-6)}=4\,\e^{-j\frac{12\pi}{5}}\e^{j\frac{2\pi}{5}n}=4\,\e^{-j\frac{2\pi}{5}}\e^{j\frac{2\pi}{5}n}$.
    \item $\sum_{k=0}^{n}a^k=\dfrac{1-a^{n+1}}{1-a}$ para $n\ge0$ y $a\ne1$, de modo que $(x\conv y)[n]=\dfrac{1-a^{n+1}}{1-a}\,u[n]$. Si $a=1$ se obtiene $(n+1)\,u[n]=r[n+1]$.
    \item Un trapecio: $0$ para $t<0$, $t$ en $[0,1]$, $1$ en $[1,2]$, $3-t$ en $[2,3]$ y $0$ para $t>3$. La duración es la suma de las duraciones.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[serie de Fourier]
  Calcula la serie de Fourier de las siguientes señales de periodo $T_0$ y comprueba el teorema de Parseval.
  \begin{enumerate}[label=\alph*)]
    \item La onda cuadrada par que vale $1$ en $\abs{t}<T_0/4$ y $-1$ en $T_0/4<\abs{t}<T_0/2$.
    \item La onda triangular par $x(t)=1-4\abs{t}/T_0$ para $\abs{t}\le T_0/2$.
  \end{enumerate}
  ¿Por qué los coeficientes de la segunda decaen más deprisa?
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $c_0=0$ y $c_k=\sinc(k/2)$: nulos para $k$ par y $\pm2/(\pi k)$ para $k$ impar. Por tanto $x(t)=\frac4\pi\big(\cos\omega_0t-\frac13\cos3\omega_0t+\frac15\cos5\omega_0t-\cdots\big)$. Parseval: $P=1=2\sum_{k\text{ impar}}\frac{4}{\pi^2k^2}$, porque $\sum_{k\text{ impar}}1/k^2=\pi^2/8$.
    \item $c_0=0$ y, por ser par, $c_k=\frac{2}{T_0}\int_0^{T_0/2}\big(1-\tfrac{4t}{T_0}\big)\cos(k\omega_0t)\dd t=\dfrac{2\big(1-(-1)^k\big)}{\pi^2k^2}$: nulos para $k$ par y $4/(\pi^2k^2)$ para $k$ impar. Así, $x(t)=\frac{8}{\pi^2}\big(\cos\omega_0t+\frac19\cos3\omega_0t+\frac1{25}\cos5\omega_0t+\cdots\big)$. Parseval: $P=\frac{1}{T_0}\int x^2\dd t=\frac13$ y $2\sum_{k\text{ impar}}\frac{16}{\pi^4k^4}=\frac{32}{\pi^4}\cdot\frac{\pi^4}{96}=\frac13$.
  \end{enumerate}
  La onda triangular es continua (su derivada es una onda cuadrada) y sus coeficientes decaen como $1/k^2$; la cuadrada tiene saltos y decae como $1/k$. Cuanto más suave es la señal, más rápido decae su espectro. De hecho, la derivada de la triangular es $-\frac{4}{T_0}$ por la cuadrada retrasada un cuarto de periodo, y los coeficientes $jk\omega_0c_k$ de la derivada coinciden con los de esa señal.
\end{solucion}

\begin{ejercicio}[transformada de Fourier]
  Calcula la transformada de Fourier de las siguientes señales, en su versión continua y discreta.
  \begin{enumerate}[label=\alph*)]
    \item Impulso: $\delta(t-t_0)$ y $\delta[n-n_0]$.
    \item Pulso rectangular, centrado y causal: $\rect(t/T)$ y $\rect\big((t-T/2)/T\big)$; en discreto, el pulso centrado de $2N+1$ muestras y el causal $u[n]-u[n-L]$.
    \item Pulso triangular, centrado y causal: $\tri(t/T)$ (duración $2T$) y $\tri\big((t-T)/T\big)$; en discreto, $t_L[n]=L-\abs{n}$ para $\abs{n}<L$ y su versión causal $t_L[n-L+1]$. \emph{Pista}: usa que el pulso triangular es la convolución de dos rectangulares.
  \end{enumerate}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $\e^{-j\omega t_0}$ y $\e^{-j\Omega n_0}$: módulo $1$ en todas las frecuencias y fase lineal.
    \item $\rect(t/T)\parF T\sinc\big(\tfrac{\omega T}{2\pi}\big)$, y el causal añade el factor $\e^{-j\omega T/2}$. En discreto, el centrado da $\sum_{n=-N}^{N}\e^{-j\Omega n}=\dfrac{\sin\big(\Omega(2N+1)/2\big)}{\sin(\Omega/2)}$, real y par; el causal, $\e^{-j\Omega\frac{L-1}{2}}\dfrac{\sin(\Omega L/2)}{\sin(\Omega/2)}$.
    \item $\tri(t/T)=\frac1T\big(\rect(\cdot/T)\conv\rect(\cdot/T)\big)(t)$, así que $\tri(t/T)\parF T\sinc^2\big(\tfrac{\omega T}{2\pi}\big)$, y el causal añade $\e^{-j\omega T}$. En discreto, $t_L$ es la convolución del pulso de $L$ muestras con su reflejado, y su transformada es $\Big(\dfrac{\sin(\Omega L/2)}{\sin(\Omega/2)}\Big)^2$; la versión causal añade $\e^{-j\Omega(L-1)}$.
  \end{enumerate}
  En los dos casos la transformada discreta es periódica de periodo $2\pi$ y, cerca de $\Omega=0$, se parece a la continua.
\end{solucion}

\begin{ejercicio}[exponenciales discretas]
  Determina si las siguientes secuencias son periódicas y, si lo son, su periodo fundamental. Para cada sinusoide, indica su frecuencia equivalente en $[-\pi,\pi)$.
  \begin{incisos}[2]
    \item $x[n]=2\,\e^{j(2\pi/8)n}$
    \item $x[n]=5\,\e^{j(3/4)n}$
    \item $x[n]=\cos(2\pi n)$
    \item $x[n]=1/n$ para $n\ne0$, con $x[0]=0$
    \item $x[n]=\cos(3\pi n/8)+\sin(\pi n/6)$
    \item $x[n]=\cos(11\pi n/6)$
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $F_0=1/8$: periódica con $N_0=8$; $\Omega_0=\pi/4$.
    \item $F_0=3/(8\pi)$ es irracional: no es periódica; $\Omega_0=0{,}75$ ya está en $[-\pi,\pi)$.
    \item $\cos(2\pi n)=1$ para todo $n$: es constante, periódica con $N_0=1$, y su frecuencia equivalente es $\Omega_0=0$.
    \item Aperiódica (e impar).
    \item El primer término tiene $F=3/16$ y periodo $16$; el segundo, $F=1/12$ y periodo $12$. La suma es periódica con $N_0=\operatorname{mcm}(16,12)=48$.
    \item $11\pi/6-2\pi=-\pi/6$, así que $x[n]=\cos(-\pi n/6)=\cos(\pi n/6)$: periódica con $N_0=12$. Aunque $11\pi/6$ parezca una frecuencia alta, la secuencia oscila despacio.
  \end{enumerate}
\end{solucion}

\section*{Sistemas}

\begin{ejercicio}[propiedades de los sistemas]
  Estudia si los siguientes sistemas tienen memoria y si son causales, estables, lineales, invariantes en el tiempo e invertibles. Para los que sean LTI, da su respuesta al impulso.
  \begin{incisos}[3]
    \item $y(t)=x(t-2)$
    \item $y[n]=x[n]+2$
    \item $y(t)=x(t)\cos(\omega_0t)$
    \item $y[n]=\sum_{k=n-2}^{n+2}x[k]$
    \item $y[n]=x[2n]$
    \item $y(t)=\e^{x(t)}$
    \item $y[n]=x[n]\,x[n-1]$
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{center}
    \small
    \begin{tabular}{@{}lcccccc@{}}
      \toprule
      & Memoria & Causal & Estable & Lineal & Invariante & Invertible\\\midrule
      a) & sí & sí & sí & sí & sí & sí\\
      b) & no & sí & sí & no & sí & sí\\
      c) & no & sí & sí & sí & no & no\\
      d) & sí & no & sí & sí & sí & no\\
      e) & sí & no & sí & sí & no & no\\
      f) & no & sí & sí & no & sí & sí\\
      g) & sí & sí & sí & no & sí & no\\\bottomrule
    \end{tabular}
  \end{center}
  \begin{enumerate}[label=\alph*)]
    \item Retardo puro: $h(t)=\delta(t-2)$; el inverso es el adelanto $x(t)=y(t+2)$.
    \item No es lineal porque la entrada nula da $y=2$. Es invertible: $x=y-2$.
    \item El modulador es lineal, pero no invariante: la respuesta a $x(t-t_0)$ es $x(t-t_0)\cos(\omega_0t)$ y no $x(t-t_0)\cos(\omega_0(t-t_0))$. No es invertible porque en los instantes en que $\cos(\omega_0t)=0$ se pierde el valor de la entrada.
    \item Media (sin normalizar) de cinco muestras centrada: usa $x[n+1]$ y $x[n+2]$, así que no es causal. Es LTI con $h[n]=u[n+2]-u[n-3]$, y estable porque $\sum\abs{h}=5$. No es invertible: la entrada $\cos(2\pi n/5)$ da salida nula, porque cinco muestras consecutivas de esa sinusoide suman cero.
    \item El diezmado: $y[1]=x[2]$ usa una entrada futura. No es invariante: si la entrada se retrasa una muestra, la salida no se retrasa (se toman las muestras impares en lugar de las pares). Pierde la mitad de las muestras: no es invertible.
    \item No es lineal ($\e^{2x}\ne2\e^{x}$). Es estable: si $\abs{x}\le B$, $y\le\e^B$. Es invertible: $x=\ln y$, porque $y>0$.
    \item No es lineal (al duplicar la entrada la salida se multiplica por cuatro) ni invertible ($x$ y $-x$ dan la misma salida). Es estable: $\abs{y}\le B^2$.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[estabilidad de sistemas continuos]
  Determina la estabilidad de los siguientes sistemas continuos causales.
  \begin{incisos}[3]
    \item $H(s)=\dfrac1s$
    \item $H(s)=\dfrac{s+2}{s+4}$
    \item $H(s)=\dfrac{1}{s-3}$
    \item $H(s)=\dfrac{s+1}{s^2+25}$
    \item $H(s)=\dfrac{s-1}{s^2+2s+5}$
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item Polo en $s=0$: no es BIBO estable. Es el integrador: el escalón produce una rampa.
    \item Polo en $s=-4$: estable.
    \item Polo en $s=3$: inestable, $h(t)=\e^{3t}u(t)$.
    \item Polos en $s=\pm5j$, sobre el eje imaginario: no es BIBO estable. $h(t)$ es una oscilación mantenida y una entrada de frecuencia $5$\,rad/s produce una salida que crece linealmente.
    \item Polos en $s=-1\pm2j$: estable. El cero en $s=1$, en el semiplano derecho, no afecta a la estabilidad.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[estabilidad de sistemas discretos]
  Determina la estabilidad de los siguientes sistemas discretos causales.
  \begin{incisos}[3]
    \item $H(z)=\dfrac{z}{z-1{,}5}$
    \item $H(z)=\dfrac{z}{(z-1)^2}$
    \item $H(z)=\dfrac{z}{z-0{,}9}$
    \item $H(z)=\dfrac{z}{z^2+1}$
    \item $H(z)=\dfrac{z-2}{z^2-z+0{,}5}$
  \end{incisos}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item Polo en $z=1{,}5$, fuera de la circunferencia unidad: inestable, $h[n]=1{,}5^nu[n]$.
    \item Polo doble en $z=1$: inestable, $h[n]=n\,u[n]$.
    \item Polo en $z=0{,}9$: estable, $h[n]=0{,}9^nu[n]$.
    \item Polos en $z=\pm j$, sobre la circunferencia unidad: no es BIBO estable. $h[n]=\sin(\pi n/2)\,u[n]$.
    \item Polos en $z=0{,}5\pm0{,}5j$, con $\abs{z}=0{,}707$: estable. El cero en $z=2$ no influye.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[respuesta acotada]
  Determina si la respuesta de cada sistema a la entrada indicada (aplicada desde $t=0$ o $n=0$) es acotada.
  \begin{enumerate}[label=\alph*)]
    \item $H(s)=\dfrac{s}{s+4}$, \ $x(t)=\sin(4t)\,u(t)$
    \item $H(s)=\dfrac{s^2}{s^2+16}$, \ $x(t)=\big(\cos(2t)+\sin(4t)\big)\,u(t)$
    \item $H(z)=\dfrac{z\sin(2\pi/7)}{z^2-2z\cos(2\pi/7)+1}$, \ $x[n]=3\cos\big(-\tfrac{4\pi}{14}n\big)\,u[n]$
    \item $H(z)=\dfrac{z}{z^2+2}$, \ $x[n]=0{,}9^nu[n]$
  \end{enumerate}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item Sí: el sistema es estable (polo en $-4$) y la entrada está acotada. En régimen permanente, $H(j4)=\frac{4j}{4+4j}=\frac{1+j}{2}$, y $y(t)\to\frac{1}{\sqrt2}\sin(4t+\pi/4)$.
    \item No. Los polos $\pm4j$ están en el eje imaginario y la entrada contiene justamente la frecuencia $4$\,rad/s: $Y(s)$ tiene polos dobles en $\pm4j$ y la salida contiene un término $t\cos(4t)$ que crece sin límite (resonancia). La parte de $\cos(2t)$ da una respuesta acotada.
    \item No. $H(z)$ es la transformada de $\sin(2\pi n/7)u[n]$, con polos $\e^{\pm j2\pi/7}$ sobre la circunferencia unidad, y la entrada es $3\cos(2\pi n/7)$ (el coseno es par), de la misma frecuencia: hay resonancia y la salida crece linealmente con $n$.
    \item No. Los polos $\pm j\sqrt2$ tienen módulo $1{,}41>1$: el sistema es inestable y la salida crece como $(\sqrt2)^n$ para esta entrada, cuya transformada no tiene ceros que cancelen esos polos.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[respuesta de sistemas continuos]
  Obtén la respuesta de los siguientes sistemas a las entradas indicadas, todas nulas para $t<0$. ¿En qué casos se puede usar la transformada de Fourier y en cuáles es necesaria la de Laplace?
  \begin{enumerate}[label=\alph*)]
    \item $h(t)=\e^{-2t}u(t)$, \ $x(t)=\big(4\cos(3t)+10\sin(5t)\big)\,u(t)$
    \item $h(t)=\sin(4t)\,u(t)$, \ $X(s)=\dfrac{s}{s^2+5}$
    \item $H(s)=\dfrac{s+3}{s+10}$, \ $x(t)=t^2u(t)$
  \end{enumerate}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $H(s)=\frac{1}{s+2}$. Fourier da el régimen permanente: con $H(j\omega)=\frac{1}{2+j\omega}$,
    \[
      y_{\mathrm{ss}}(t)=\frac{4}{\sqrt{13}}\cos\big(3t-\arctan\tfrac32\big)+\frac{10}{\sqrt{29}}\sin\big(5t-\arctan\tfrac52\big).
    \]
    Laplace da además el transitorio, un término $A\e^{-2t}$ con $A=-y_{\mathrm{ss}}(0)=-\big(\frac{8}{13}-\frac{50}{29}\big)=\frac{418}{377}\approx1{,}109$, de modo que $y(t)=\big(\frac{418}{377}\e^{-2t}+y_{\mathrm{ss}}(t)\big)u(t)$.
    \item $H(s)=\frac{4}{s^2+16}$ y $Y(s)=\frac{4s}{(s^2+5)(s^2+16)}=\frac{4}{11}\Big(\frac{s}{s^2+5}-\frac{s}{s^2+16}\Big)$, de donde $y(t)=\frac{4}{11}\big(\cos(\sqrt5\,t)-\cos(4t)\big)u(t)$. Es acotada porque la frecuencia de la entrada, $\sqrt5$, no coincide con la del polo, $4$; pero el sistema no es estable y Fourier no se puede aplicar.
    \item $Y(s)=\frac{2(s+3)}{s^3(s+10)}=\frac{3}{5s^3}+\frac{7}{50s^2}-\frac{7}{500s}+\frac{7}{500(s+10)}$, y
    \[
      y(t)=\Big(\frac{3}{10}t^2+\frac{7}{50}t-\frac{7}{500}+\frac{7}{500}\e^{-10t}\Big)u(t).
    \]
    La entrada $t^2u(t)$ no tiene transformada de Fourier: hace falta Laplace.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[respuesta de sistemas discretos]
  Obtén la respuesta de los siguientes sistemas a las entradas indicadas, todas nulas para $n<0$.
  \begin{enumerate}[label=\alph*)]
    \item $H(z)=\dfrac{z}{z+0{,}8}$, \ $x[n]=\big(0{,}5^n+4\cdot0{,}8^{n-1}\big)\,u[n]$
    \item $H(z)=\dfrac{z}{z-0{,}8}$, \ $x[n]=n\,u[n]$
    \item $H(z)=\dfrac{z^2}{z^2-0{,}3z+0{,}4}$, \ $x[n]=\cos\big(\tfrac{\pi}{6}n\big)\,u[n]$. Obtén al menos el régimen permanente con la respuesta en frecuencia y la forma del transitorio.
  \end{enumerate}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $X(z)=\frac{z}{z-0{,}5}+5\,\frac{z}{z-0{,}8}$. Descomponiendo $Y(z)/z$ en fracciones simples,
    \[
      y[n]=\Big(\tfrac{81}{26}(-0{,}8)^n+\tfrac52(0{,}8)^n+\tfrac{5}{13}(0{,}5)^n\Big)u[n].
    \]
    Comprobación: $y[0]=6=x[0]$ y $y[1]=-0{,}3$, igual que con \texttt{lfilter([1], [1, 0.8], x)}.
    \item $Y(z)=\frac{z}{z-0{,}8}\cdot\frac{z}{(z-1)^2}=\frac{20z}{z-0{,}8}-\frac{20z}{z-1}+\frac{5z}{(z-1)^2}$, de donde $y[n]=\big(20\cdot0{,}8^n-20+5n\big)u[n]$, con $y[0]=0$, $y[1]=1$, $y[2]=2{,}8$, \ldots Crece porque la entrada crece; la ganancia en continua es $H(1)=5$.
    \item Los polos $0{,}15\pm0{,}614j$ tienen módulo $\sqrt{0{,}4}=0{,}632<1$: el sistema es estable. Con $H(\e^{j\Omega})=\frac{1}{1-0{,}3\e^{-j\Omega}+0{,}4\e^{-j2\Omega}}$, en $\Omega=\pi/6$ se obtiene $\abs{H}=1{,}041$ y $\arg H=0{,}206$\,rad, así que
    \[
      y[n]=1{,}041\cos\big(\tfrac\pi6n+0{,}206\big)+0{,}632^n\big(A\cos(1{,}331n)+B\sin(1{,}331n)\big),
    \]
    donde $1{,}331=\arg(0{,}15+0{,}614j)$. Imponiendo $y[0]=1$ e $y[1]=1{,}166$ (de la recursión) resulta $A=-0{,}019$ y $B=0{,}639$. El transitorio se reduce a menos del 1\,\% en unas diez muestras.
  \end{enumerate}
\end{solucion}

\begin{ejercicio}[régimen permanente]
  Obtén con la transformada de Fourier la respuesta en régimen permanente de los siguientes sistemas a las entradas indicadas.
  \begin{enumerate}[label=\alph*)]
    \item $H(s)=\dfrac{1}{s+4}$, \ $x(t)$ la onda cuadrada del ejercicio~7 con $T_0=2$\,s.
    \item $H(z)=\dfrac{z}{z-0{,}8}$, \ $x[n]=2\cos\big(\tfrac\pi6n\big)+\sin\big(\tfrac\pi2n\big)$.
    \item $H(z)=\dfrac{z}{z-1{,}2}$, \ $x[n]=\cos\big(\tfrac{2\pi}{10}n\big)$.
  \end{enumerate}
\end{ejercicio}
\begin{solucion}
  \begin{enumerate}[label=\alph*)]
    \item $\omega_0=\pi$\,rad/s. Cada armónico impar, de amplitud $\frac{4}{\pi k}(-1)^{(k-1)/2}$, se multiplica por $\abs{H(jk\pi)}=1/\sqrt{16+k^2\pi^2}$ y se desfasa $-\arctan(k\pi/4)$:
    \[
      y_{\mathrm{ss}}(t)=0{,}250\cos(\pi t-0{,}666)-0{,}041\cos(3\pi t-1{,}169)+0{,}016\cos(5\pi t-1{,}321)-\cdots
    \]
    El paso bajo atenúa más los armónicos altos: la salida se parece mucho más a una sinusoide que la entrada.
    \item $H(\e^{j\Omega})=\frac{1}{1-0{,}8\e^{-j\Omega}}$. En $\pi/6$: $\abs{H}=1{,}983$, $\arg H=-0{,}916$; en $\pi/2$: $\abs{H}=0{,}781$, $\arg H=-0{,}675$. Por tanto
    \[
      y_{\mathrm{ss}}[n]=3{,}966\cos\big(\tfrac\pi6n-0{,}916\big)+0{,}781\sin\big(\tfrac\pi2n-0{,}675\big).
    \]
    \item El polo $z=1{,}2$ está fuera de la circunferencia unidad: el sistema es inestable, la salida crece como $1{,}2^n$ y no hay régimen permanente. La respuesta en frecuencia $H(\e^{j\Omega})$ se puede evaluar, pero no describe la respuesta del sistema.
  \end{enumerate}
\end{solucion}
